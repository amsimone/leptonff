cmake_minimum_required(VERSION 3.16)
project(leptonff_fortran_example LANGUAGES CXX Fortran)
find_package(leptonff 0.27 CONFIG REQUIRED)
if(TARGET leptonff::fortran)
  set(fortran_interface leptonff::fortran)
else()
  # Portable fallback when the library was installed without a Fortran compiler.
  add_library(leptonff_fortran_module "${leptonff_FORTRAN_SOURCE}")
  target_link_libraries(leptonff_fortran_module PUBLIC leptonff::leptonff)
  set_target_properties(leptonff_fortran_module PROPERTIES Fortran_MODULE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/modules")
  target_include_directories(leptonff_fortran_module INTERFACE "${CMAKE_CURRENT_BINARY_DIR}/modules")
  set(fortran_interface leptonff_fortran_module)
endif()
add_executable(decay_fortran main.f90)
target_link_libraries(decay_fortran PRIVATE ${fortran_interface})
