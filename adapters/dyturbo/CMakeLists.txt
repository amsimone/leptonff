cmake_minimum_required(VERSION 3.20)
project(leptonff_dyturbo LANGUAGES CXX)
set(DYTURBO_ROOT "" CACHE PATH "DYTurbo 1.4.2 installation prefix")
if(NOT DYTURBO_ROOT)
  message(FATAL_ERROR "Set DYTURBO_ROOT to the DYTurbo 1.4.2 installation prefix")
endif()
execute_process(COMMAND "${DYTURBO_ROOT}/bin/dyturbo-config" --version
  OUTPUT_VARIABLE DYTURBO_VERSION OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE config_status)
if(NOT config_status EQUAL 0 OR NOT DYTURBO_VERSION STREQUAL "1.4.2")
  message(FATAL_ERROR "This native client requires DYTurbo 1.4.2")
endif()
# Reconfiguring against another prefix must not retain a previous backend.
unset(DYTURBO_INCLUDE_DIR CACHE)
unset(DYTURBO_LIBRARY CACHE)
find_path(DYTURBO_INCLUDE_DIR dyturbo/dyturbo.h PATHS "${DYTURBO_ROOT}/include" NO_DEFAULT_PATH REQUIRED)
find_library(DYTURBO_LIBRARY dyturbo PATHS "${DYTURBO_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
add_executable(leptonff_dyturbo native.cpp)
target_compile_features(leptonff_dyturbo PRIVATE cxx_std_11)
target_include_directories(leptonff_dyturbo PRIVATE "${DYTURBO_INCLUDE_DIR}/dyturbo")
target_link_libraries(leptonff_dyturbo PRIVATE "${DYTURBO_LIBRARY}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/backend-paths.txt"
  "${DYTURBO_LIBRARY}\n${DYTURBO_ROOT}/share/dyturbo/default.in\n")
install(TARGETS leptonff_dyturbo RUNTIME DESTINATION bin)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/backend-paths.txt" DESTINATION bin)
option(LEPTONFF_DYTURBO_MASS_EXAMPLE "Build inclusive mass-migration example" OFF)
if(LEPTONFF_DYTURBO_MASS_EXAMPLE)
  find_package(leptonff 0.27 CONFIG REQUIRED)
  add_executable(leptonff_dyturbo_mass mass_response.cpp)
  target_link_libraries(leptonff_dyturbo_mass PRIVATE leptonff::leptonff)
  target_compile_features(leptonff_dyturbo_mass PRIVATE cxx_std_17)
endif()

option(LEPTONFF_DYTURBO_DECAY_CLIENT "Build differential DYTurbo/decay/PHOTOS client" OFF)
if(LEPTONFF_DYTURBO_DECAY_CLIENT)
  find_package(leptonff 0.27 CONFIG REQUIRED)
  set(PHOTOS_SOURCE "" CACHE PATH "PHOTOS source directory")
  set(PHOTOS_LIBRARY "" CACHE FILEPATH "PHOTOS library with matching C++ ABI")
  set(CUBA_ROOT "" CACHE PATH "Cuba headers directory")
  if(NOT IS_DIRECTORY "${PHOTOS_SOURCE}" OR NOT EXISTS "${PHOTOS_LIBRARY}" OR NOT EXISTS "${CUBA_ROOT}/cuba.h")
    message(FATAL_ERROR "Set PHOTOS_SOURCE, PHOTOS_LIBRARY and CUBA_ROOT")
  endif()
  set(photos_dirs "${PHOTOS_SOURCE}")
  file(GLOB photos_children LIST_DIRECTORIES true "${PHOTOS_SOURCE}/*")
  foreach(child IN LISTS photos_children)
    if(IS_DIRECTORY "${child}")
      list(APPEND photos_dirs "${child}")
    endif()
  endforeach()
  add_executable(leptonff_dyturbo_decay decay_native.cpp photos_decay.cpp)
  target_compile_features(leptonff_dyturbo_decay PRIVATE cxx_std_17)
  target_include_directories(leptonff_dyturbo_decay PRIVATE "${DYTURBO_ROOT}/include/dyturbo" "${CUBA_ROOT}" ${photos_dirs})
  target_link_libraries(leptonff_dyturbo_decay PRIVATE "${DYTURBO_LIBRARY}" leptonff::leptonff "${PHOTOS_LIBRARY}")
  file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/decay-backend-paths.txt"
    CONTENT "${DYTURBO_LIBRARY}\n${DYTURBO_ROOT}/share/dyturbo/default.in\n${PHOTOS_LIBRARY}\n$<TARGET_FILE:leptonff::leptonff>\n")
  add_executable(leptonff_dyturbo_mass_precision mass_precision.cpp photos_decay.cpp)
  target_compile_features(leptonff_dyturbo_mass_precision PRIVATE cxx_std_17)
  target_include_directories(leptonff_dyturbo_mass_precision PRIVATE ${photos_dirs})
  target_link_libraries(leptonff_dyturbo_mass_precision PRIVATE leptonff::leptonff "${PHOTOS_LIBRARY}")
endif()
